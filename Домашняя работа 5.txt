double var_16(double x, double y, double z);

int main()
{
	setlocale(LC_ALL, "RUS"); 
	double x, y, z;
	double result;
	puts("Введите первое число");
	scanf("%lf", &x);
	puts("Введите второе число");
	scanf("%lf", &y);
	puts("Введите третье число");
	scanf("%lf", &z);
	printf("x = %lf, y = %lf, z = %lf\n", x, y, z);
	result = var_16(x, y, z);
	printf("w = %lf\n", result);
	return 0;
}

double var_16(double x, double y, double z)
{
        double res, res_1, res_2, res_3;
        res_1 = atan(x) + atan(z);
        res_2 = exp(fabs(x - y)) * pow(fabs(x - y), x + y);
        res_3 = pow((pow(x, 6) + pow(log(y), 2)), 1./3);
        res = res_3 + (res_2 / res_1);
	return res;
}
